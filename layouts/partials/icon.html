{{- /* Simple, fileless icon renderer. * If a known name is passed, render an inline SVG. * Otherwise, try to load from assets via resources.Get with safe fallbacks. */ -}} {{- $name := .name | default "" -}} {{- $class := .class | default "h-6 w-6" -}} {{- if not $name -}} ⬜ {{- else -}} {{- /* 1) Inline SVG for common names (no asset files needed) */ -}} {{- if eq $name "book" -}} {{- else if eq $name "microphone" -}} {{- else if eq $name "sparkles" -}} {{- else if eq $name "code-bracket" -}} {{- else if eq $name "star" -}} {{- else if or (eq $name "rectangle-group") (eq $name "grid") -}} {{- else if or (eq $name "brands/github") (eq $name "github") -}} {{- else if or (eq $name "brands/instagram") (eq $name "instagram") -}} {{- else if or (eq $name "at-symbol") (eq $name "email") -}} {{- else -}} {{- /* 2) 마지막 시도: 로컬 assets에서 찾아보기 (없으면 ⬜ 출력) */ -}} {{- $cands := slice (printf "media/icons/%s.svg" $name) (printf "icons/%s.svg" $name) (printf "media/icons/hero/outline/%s.svg" $name) (printf "media/icons/hero/solid/%s.svg" $name) (printf "media/icons/brands/%s.svg" $name) (printf "icons/hero/outline/%s.svg" $name) (printf "icons/hero/solid/%s.svg" $name) (printf "icons/brands/%s.svg" $name) -}} {{- $.Scratch.Delete "hb_icon_svg" -}} {{- range $cands -}} {{- with resources.Get . -}} {{- $.Scratch.Set "hb_icon_svg" . -}} {{- break -}} {{- end -}} {{- end -}} {{- with $.Scratch.Get "hb_icon_svg" -}} {{ .Content | safeHTML }} {{- else -}} ⬜ {{- end -}} {{- end -}} {{- end -}}
