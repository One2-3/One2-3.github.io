"brands/github" "class" "h-6 w-6") }} 동작: 1) name 에 경로가 포함되어 있으면 그 경로를 우선 시도 (예: hero/outline/book-open) 2) 아니면 hero/outline, hero/solid, brands, academicons 순으로 시도 3) 그래도 못 찾으면 media/icons/**/.svg 를 와일드카드로 탐색 4) 그래도 없으면 ⬜ fallback */}} {{- $name := .name | default "" -}} {{- $class := .class | default "h-6 w-6" -}} {{- if not $name -}} ⬜ {{- else -}} {{- $candidates := slice -}} {{/* 1) name 이 경로 형태면 그대로 후보에 추가 */}} {{- if or (strings.Contains $name "/") (strings.HasSuffix $name ".svg") -}} {{- $clean := strings.TrimSuffix $name ".svg" -}} {{- $candidates = $candidates | append (printf "media/icons/hero/outline/%s.svg" $name) -}} {{- $candidates = $candidates | append (printf "media/icons/hero/solid/%s.svg" $name) -}} {{- $candidates = $candidates | append (printf -}} {{- $candidates = $candidates | append (printf "icons/academicons/%s.svg" $name) -}} {{- end -}} {{/* 후보 경로에서 첫 발견 리소스 선택 */}} {{- $svg := nil -}} {{- range $candidates -}} {{- if not $svg -}} {{- -}} {{- if $hits -}} {{- $svg = index $hits 0 -}} {{- end -}} {{- end -}} {{- if $svg -}} {{ $svg.Content | safeHTML }} {{- else -}} {{/* 마지막 안전장치 */}} ⬜ {{- end -}} {{- end -}}
