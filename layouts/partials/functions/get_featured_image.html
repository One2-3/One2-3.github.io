{{/* 안전판: 문자열/객체 모두 지원 + Hugo Resource 우선 반환 */}} {{- $p := . -}} {{- $imgParam := $p.Params.image -}} {{- $name := "" -}} {{- if $imgParam -}} {{- /* 타입 판별을 printf "%T" 로 처리 (reflect 불사용) */ -}} {{- $t := printf "%T" $imgParam -}} {{- if in $t "map" -}} {{- $name = or (index $imgParam "filename") (index $imgParam "src") (index $imgParam "url") "" -}} {{- else -}} {{- $name = $imgParam -}} {{- end -}} {{- end -}} {{- $res := nil -}} {{- if $name -}} {{- /* 1) 페이지 번들에서 */ -}} {{- $res = $p.Resources.GetMatch $name -}} {{- /* 2) assets/media/ 에서 */ -}} {{- if not $res -}}{{- $res = resources.Get (printf "media/%s" $name) -}}{{- end -}} {{- /* 3) assets 루트에서 직접 (사용자가 media/ 로 적었을 때) */ -}} {{- if and (not $res) (hasPrefix $name "media/") -}} {{- $res = resources.Get $name -}} {{- end -}} {{- end -}} {{- if not $res -}} {{- /* 4) 번들 첫 이미지 fallback */ -}} {{- $res = ($p.Resources.ByType "image" | first) -}} {{- end -}} {{- return $res -}}
