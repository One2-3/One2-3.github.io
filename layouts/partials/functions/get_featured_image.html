{{/* Featured image resolver (초안전) - .Params.image: "pro1.jpg" 같은 문자열 또는 {filename/src/url: "..."} 맵 모두 지원 (maps.Params 포함) - .Resources 유무와 무관하게 안전 - 우선순위: 번들(.Resources.GetMatch) → assets/media → assets 루트의 media/... → 번들 첫 이미지 - 반환: Hugo Resource (없으면 nil) */}} {{- $p := . -}} {{- $imgParam := $p.Params.image -}} {{/* 1) 이미지 이름 문자열 추출 */}} {{- $name := "" -}} {{- if $imgParam -}} {{- $t := printf "%T" $imgParam -}} {{- $isMap := or (in $t "map") (in $t "Params") -}} {{- if $isMap -}} {{- $name = or (index $imgParam "filename") (index $imgParam "src") (index $imgParam "url") "" -}} {{- else -}} {{- $name = $imgParam -}} {{- end -}} {{- end -}} {{/* 2) 후보 경로/리소스 단계별로 구함 (서로 다른 변수에만 담음) */}} {{- $resFromPage := nil -}} {{- with $p.Resources -}} {{- if and $name (ne $name "") -}} {{- $resFromPage = .GetMatch $name -}} {{- end -}} {{- end -}} {{- $resFromAssets := nil -}} {{- if and (not $resFromPage) (and $name (ne $name "")) -}} {{- $resFromAssets = resources.Get (printf "media/%s" $name) -}} {{- end -}} {{- $resFromAssetsDirect := nil -}} {{- if and (not $resFromPage) (not $resFromAssets) (and $name (ne $name "")) (hasPrefix $name "media/") -}} {{- $resFromAssetsDirect = resources.Get $name -}} {{- end -}} {{- $resFromBundleFirst := nil -}} {{- with $p.Resources -}} {{- if not (or $resFromPage $resFromAssets $resFromAssetsDirect) -}} {{- $resFromBundleFirst = ( .ByType "image" | first ) -}} {{- end -}} {{- end -}} {{/* 3) 최종 선택: 첫 번째 non-nil */}} {{- $res := or $resFromPage (or $resFromAssets (or $resFromAssetsDirect $resFromBundleFirst)) -}} {{- return $res -}}
