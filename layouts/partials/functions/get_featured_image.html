{{/* 안전판 featured image resolver - .Params.image 가 string 또는 map(filename/src/url) 모두 지원 - page.Resources 가 없는 페이지에서도 안전하게 동작 (가드 포함) - 가능하면 Hugo Resource(이미지)를 반환, 없으면 nil 반환 */}} {{- $p := . -}} {{- $imgParam := $p.Params.image -}} {{- $name := "" -}} {{- if $imgParam -}} {{- /* 타입 문자열화 (reflect 미사용) */ -}} {{- $t := printf "%T" $imgParam -}} {{- $isMap := or (in $t "map") (in $t "Params") -}} {{- if $isMap -}} {{- $name = or (index $imgParam "filename") (index $imgParam "src") (index $imgParam "url") "" -}} {{- else -}} {{- $name = $imgParam -}} {{- end -}} {{- end -}} {{- $res := nil -}} {{- if and $name (ne $name "") -}} {{- /* 1) 페이지 번들 내부 (Resources가 있을 때만) */ -}} {{- with $p.Resources -}} {{- $res = .GetMatch $name -}} {{- end -}} {{- /* 2) assets/media/ */ -}} {{- if not $res -}} {{- $res = resources.Get (printf "media/%s" $name) -}} {{- end -}} {{- /* 3) assets 루트에서 직접 (사용자가 media/...로 적은 경우) */ -}} {{- if and (not $res) (hasPrefix $name "media/") -}} {{- $res = resources.Get $name -}} {{- end -}} {{- end -}} {{- if not $res -}} {{- /* 4) 번들 첫 이미지 (Resources가 있을 때만) */ -}} {{- with $p.Resources -}} {{- $res = (.ByType "image" | first) -}} {{- end -}} {{- end -}} {{- return $res -}}
