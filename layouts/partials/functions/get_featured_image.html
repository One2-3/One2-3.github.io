{{/* Featured image resolver (초-보수 버전) - 오직 페이지 번들(.Resources) 안에서만 대표 이미지 탐색 - .Params.image: "pro1.jpg" 또는 {filename/src/url: "..."} 모두 처리 - 글로벌 resources.Get 호출을 전부 제거 → "nil is not a command" 회피 - 반환: Hugo Resource (없으면 nil) */}} {{- $p := . -}} {{- $imgParam := $p.Params.image -}} {{/* 1) 이미지 파일명만 추출 (URL/절대경로는 여기선 미처리) */}} {{- $name := "" -}} {{- if $imgParam -}} {{- $t := printf "%T" $imgParam -}} {{- $isMap := or (in $t "map") (in $t "Params") -}} {{- if $isMap -}} {{- /* filename/src/url 중 filename 우선. URL/절대경로는 번들 탐색 대상이 아님 */ -}} {{- $name = or (index $imgParam "filename") (index $imgParam "src") (index $imgParam "url") "" -}} {{- else -}} {{- $name = $imgParam -}} {{- end -}} {{- end -}} {{/* 2) 번들 내부에서만 탐색 (외부/정적 경로는 호출하지 않음) */}} {{- $res := nil -}} {{- with $p.Resources -}} {{- if and $name (ne $name "") -}} {{- $res = .GetMatch $name -}} {{- end -}} {{- if not $res -}} {{- $res = ( .ByType "image" | first ) -}} {{- end -}} {{- end -}} {{- return $res -}}
