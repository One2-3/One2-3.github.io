{{/* 안전판: 문자열/객체(image.filename/src/url) 모두 지원 + .Resources nil 가드 + Resource 우선 */}} {{- $p := . -}} {{- $imgParam := $p.Params.image -}} {{- $name := "" -}} {{- if $imgParam -}} {{- $t := printf "%T" $imgParam -}} {{- if in $t "map" -}} {{- $name = or (index $imgParam "filename") (index $imgParam "src") (index $imgParam "url") "" -}} {{- else -}} {{- $name = $imgParam -}} {{- end -}} {{- end -}} {{- $res := nil -}} {{- if $name -}} {{- /* 1) 페이지 번들 내부에서 (Resources가 있을 때만) */ -}} {{- with $p.Resources -}} {{- $res = .GetMatch $name -}} {{- end -}} {{- /* 2) assets/media/ */ -}} {{- if not $res -}}{{- $res = resources.Get (printf "media/%s" $name) -}}{{- end -}} {{- /* 3) assets 루트에서 직접 (사용자가 media/로 적은 경우) */ -}} {{- if and (not $res) (hasPrefix $name "media/") -}} {{- $res = resources.Get $name -}} {{- end -}} {{- end -}} {{- if not $res -}} {{- /* 4) 번들 첫 이미지 (Resources가 있을 때만) */ -}} {{- with $p.Resources -}} {{- $res = (.ByType "image" | first) -}} {{- end -}} {{- end -}} {{- return $res -}}
