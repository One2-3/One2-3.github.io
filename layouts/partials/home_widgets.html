{{/* layouts/partials/home_widgets.html — 언어별(.Site)로 home 찾기, leaf/section 모두 지원 */}} {{- $site := .Site -}} {{/* 1) 현재 언어 사이트에서만 home을 찾는다 (모호 호출 금지) */}} {{- $home := $site.GetPage "section" "home" -}} {{/* content//home/_index.md */}} {{- if not $home -}} {{- $home = $site.GetPage "/home" -}} {{- end -}} {{/* content//home/index.md */}} {{- if not $home -}} {{- $home = $site.GetPage "page" "home/index" -}} {{- end -}} {{- if not $home -}}

위젯 섹션을 찾을 수 없습니다.

{{- else -}}
{{/* 2) 위젯 수집: 섹션이면 .Pages, 리프면 .Resources */}} {{- $widgets := slice -}} {{- if eq $home.Kind "section" -}} {{- $widgets = $home.Pages -}} {{- else -}} {{- $widgets = $home.Resources.ByType "page" -}} {{- if not $widgets -}} {{- $widgets = $home.Resources.Match "**.md" -}} {{- end -}} {{- end -}} {{/* widget 지정된 것만 weight 순 정렬 */}} {{- $widgets = where $widgets ".Params.widget" "!=" nil -}} {{- $widgets = sort $widgets "Params.weight" "asc" -}} {{- range $w := $widgets -}} {{- $wname := lower (printf "%v" (index $w.Params "widget")) -}} {{- $wfile := replace $wname "." "_" -}} {{/* about.avatar → about_avatar */}} {{- $partial := printf "widgets/%s.html" $wfile -}} {{- if templates.Exists (printf "partials/%s" $partial) -}} {{- partial $partial (dict "page" . "widget" $w) -}} {{- else -}}
알 수 없는 위젯: {{ $wname }}
{{- end -}} {{- end -}}
{{- end -}}
