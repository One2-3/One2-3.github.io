{{/* layouts/partials/home_widgets.html — 다국어 안전판 (재대입/타입지정/모호호출 없음) */}} {{- $lang := .Lang -}} {{/* 1) home 페이지를 확정: 현재 언어 컨텍스트(.Site) → 전역 site + 언어 prefix 순서로 OR 체인 */}} {{- $home := or (.Site.GetPage "/home") (.Site.GetPage "/home/index") (site.GetPage (printf "/%s/home" $lang)) (site.GetPage (printf "/%s/home/index" $lang)) -}} {{- if not $home -}}

위젯 섹션을 찾을 수 없습니다.

{{- else -}}
{{/* 2) 위젯 수집: 섹션이면 .Pages, 리프면 .Resources */}} {{- $widgets := cond (eq $home.Kind "section") $home.Pages ($home.Resources.ByType "page") -}} {{- if not $widgets -}}{{- $widgets = $home.Resources.Match "**.md" -}}{{- end -}} {{/* widget 지정된 항목만 weight 오름차순 렌더 */}} {{- $widgets = where $widgets ".Params.widget" "!=" nil -}} {{- $widgets = sort $widgets "Params.weight" "asc" -}} {{- if not (gt (len $widgets) 0) -}}

위젯 섹션을 찾을 수 없습니다.

{{- else -}} {{- range $w := $widgets -}} {{- $wname := lower (printf "%v" (index $w.Params "widget")) -}} {{- $partial := printf "widgets/%s.html" (replace $wname "." "_") -}} {{- if templates.Exists (printf "partials/%s" $partial) -}} {{- partial $partial (dict "page" . "widget" $w) -}} {{- else -}}
알 수 없는 위젯: {{ $wname }}
{{- end -}} {{- end -}} {{- end -}}
{{- end -}}
