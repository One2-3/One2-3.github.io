{{/* home_widgets.html — 섹션(Branch) 홈 전용, 다국어 안전(.Site만 사용) */}} {{- $home := .Site.GetPage "section" "home" -}} {{- if not $home -}}

위젯 섹션을 찾을 수 없습니다.

{{- else -}}
{{/* 섹션 자식(.Pages) 중 widget이 있는 것만 weight 순으로 */}} {{- $widgets := where $home.Pages ".Params.widget" "!=" nil -}} {{- $widgets = sort $widgets "Params.weight" "asc" -}} {{- if not (gt (len $widgets) 0) -}}

위젯 섹션을 찾을 수 없습니다.

{{- else -}} {{- range $w := $widgets -}} {{- $wname := lower (printf "%v" (index $w.Params "widget")) -}} {{- $partial := printf "widgets/%s.html" (replace $wname "." "_") -}} {{- if templates.Exists (printf "partials/%s" $partial) -}} {{- partial $partial (dict "page" . "widget" $w) -}} {{- else -}}
알 수 없는 위젯: {{ $wname }}
{{- end -}} {{- end -}} {{- end -}}
{{- end -}}
