{{/* layouts/partials/home_widgets.html — 언어별 home을 정확 경로로 조회, leaf/section 모두 지원 */}} {{- $lang := .Lang -}} {{/* 1) 현재 언어의 home을 '정확 경로'로만 찾기 (모호 호출 금지) */}} {{- $home := site.GetPage "page" (printf "/%s/home/index" $lang) -}} {{/* leaf */}} {{- if not $home -}} {{- $home = site.GetPage "section" (printf "/%s/home" $lang) -}} {{/* section */}} {{- end -}} {{- if not $home -}} {{- $home = site.GetPage (printf "/%s/home" $lang) -}} {{/* 마지막 폴백 */}} {{- end -}} {{- if not $home -}}

위젯 섹션을 찾을 수 없습니다.

{{- else -}}
{{/* 2) 위젯 수집: 섹션이면 .Pages, 리프면 .Resources */}} {{- $widgets := slice -}} {{- if eq $home.Kind "section" -}} {{- $widgets = $home.Pages -}} {{- else -}} {{- $widgets = $home.Resources.ByType "page" -}} {{- if not $widgets -}} {{- $widgets = $home.Resources.Match "**.md" -}} {{- end -}} {{- end -}} {{/* widget 지정된 것만 weight 순 정렬 후 렌더 */}} {{- $widgets = where $widgets ".Params.widget" "!=" nil -}} {{- $widgets = sort $widgets "Params.weight" "asc" -}} {{- range $w := $widgets -}} {{- $wname := lower (printf "%v" (index $w.Params "widget")) -}} {{- $wfile := replace $wname "." "_" -}} {{/* about.avatar → about_avatar */}} {{- $partial := printf "widgets/%s.html" $wfile -}} {{- if templates.Exists (printf "partials/%s" $partial) -}} {{- partial $partial (dict "page" . "widget" $w) -}} {{- else -}}
알 수 없는 위젯: {{ $wname }}
{{- end -}} {{- end -}}
