{{/* layouts/partials/home_widgets.html — 다국어 안전판 (모든 경로 폴백), leaf/section 모두 지원 */}} {{- $lang := .Lang -}} {{- $home := nil -}} {{/* A. 현재 언어(.Site) 컨텍스트에서 우선 시도 */}} {{- if not $home -}}{{- $home = .Site.GetPage "/home" -}}{{- end -}} {{- if not $home -}}{{- $home = .Site.GetPage "/home/index" -}}{{- end -}} {{/* B. 그래도 못 찾으면 전역 site에 언어 prefix 붙여서 재시도 */}} {{- if not $home -}}{{- $home = site.GetPage (printf "/%s/home" $lang) -}}{{- end -}} {{- if not $home -}}{{- $home = site.GetPage (printf "/%s/home/index" $lang) -}}{{- end -}} {{- if not $home -}}

위젯 섹션을 찾을 수 없습니다.

{{- else -}}
{{/* C. 위젯 수집: 섹션이면 .Pages, 리프(index.md)면 .Resources */}} {{- $widgets := slice -}} {{- if eq $home.Kind "section" -}} {{- $widgets = $home.Pages -}} {{- else -}} {{- $widgets = $home.Resources.ByType "page" -}} {{- if not $widgets -}}{{- $widgets = $home.Resources.Match "**.md" -}}{{- end -}} {{- end -}} {{/* D. widget 지정된 항목만 weight 오름차순 렌더 */}} {{- $widgets = where $widgets ".Params.widget" "!=" nil -}} {{- $widgets = sort $widgets "Params.weight" "asc" -}} {{- if not (gt (len $widgets) 0) -}}

위젯 섹션을 찾을 수 없습니다.

{{- else -}} {{- range $w := $widgets -}} {{- $wname := lower (printf "%v" (index $w.Params "widget")) -}} {{- $partial := printf "widgets/%s.html" (replace $wname "." "_") -}} {{- if templates.Exists (printf "partials/%s" $partial) -}} {{- partial $partial (dict "page" . "widget" $w) -}} {{- else -}}
알 수 없는 위젯: {{ $wname }}
{{- end -}} {{- end -}} {{- end -}}
{{- end -}}
