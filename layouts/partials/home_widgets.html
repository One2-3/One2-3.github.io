{{/* home_widgets.html — 다국어 안전판 (섹션/리프 둘 다 지원) */}} {{- $site := .Site -}} {{/* 1) 현재 언어의 "home" 페이지를 우선 섹션으로 시도, 실패 시 리프(leaf)로 폴백 */}} {{- $home := or ($site.GetPage "section" "home") ($site.GetPage "/home") ($site.GetPage "/home/index") -}} {{- if not $home -}}

위젯 섹션을 찾을 수 없습니다.

{{- else -}}
{{/* 2) 위젯 수집: 섹션이면 .Pages, 리프면 .Resources */}} {{- $widgets := slice -}} {{- if eq $home.Kind "section" -}} {{- $widgets = $home.Pages -}} {{- else -}} {{- $widgets = $home.Resources.ByType "page" -}} {{- if not $widgets -}}{{- $widgets = $home.Resources.Match "**.md" -}}{{- end -}} {{- end -}} {{/* 3) widget 프론트매터가 있는 것만 weight 오름차순 */}} {{- $widgets = where $widgets ".Params.widget" "!=" nil -}} {{- $widgets = sort $widgets "Params.weight" "asc" -}} {{- if not (gt (len $widgets) 0) -}}

위젯 섹션을 찾을 수 없습니다.

{{- else -}} {{- range $w := $widgets -}} {{- $wname := lower (printf "%v" (index $w.Params "widget")) -}} {{- $partial := printf "widgets/%s.html" (replace $wname "." "_") -}} {{- if templates.Exists (printf "partials/%s" $partial) -}} {{- partial $partial (dict "page" . "widget" $w) -}} {{- else -}}
Unknown widget: {{ $wname }}
{{- end -}} {{- end -}} {{- end -}}
{{- end -}}
