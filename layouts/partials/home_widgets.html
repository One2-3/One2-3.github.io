{{/* layouts/partials/home_widgets.html — 언어 prefix 절대경로만 사용, leaf/section 모두 지원 */}} {{- $lang := .Lang -}} {{/* 1) 현재 언어의 /home을 정확한 경로로만 조회 (모호 호출 금지) */}} {{- $home := site.GetPage (printf "/%s/home" $lang) -}} {{- if not $home -}}{{- $home = site.GetPage (printf "/%s/home/index" $lang) -}}{{- end -}} {{- if not $home -}}

위젯 섹션을 찾을 수 없습니다.

{{- else -}}
{{/* 2) 위젯 수집: 섹션이면 .Pages, 리프(index.md)면 .Resources */}} {{- $widgets := slice -}} {{- if eq $home.Kind "section" -}} {{- $widgets = $home.Pages -}} {{- else -}} {{- $widgets = $home.Resources.ByType "page" -}} {{- if not $widgets -}}{{- $widgets = $home.Resources.Match "**.md" -}}{{- end -}} {{- end -}} {{/* widget 지정된 항목만 weight 오름차순 렌더 */}} {{- $widgets = where $widgets ".Params.widget" "!=" nil -}} {{- $widgets = sort $widgets "Params.weight" "asc" -}} {{- range $w := $widgets -}} {{- $wname := lower (printf "%v" (index $w.Params "widget")) -}} {{- $partial := printf "widgets/%s.html" (replace $wname "." "_") -}} {{- if templates.Exists (printf "partials/%s" $partial) -}} {{- partial $partial (dict "page" . "widget" $w) -}} {{- else -}}
알 수 없는 위젯: {{ $wname }}
{{- end -}} {{- end -}}
{{- end -}}
