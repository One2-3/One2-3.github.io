{{/* layouts/partials/home_widgets.html — .Site 컨텍스트에서만 home 찾기 (leaf/section 모두 지원) */}} {{- /* 1) 현재 언어(.Site)에서만 home을 찾는다 */ -}} {{- $home := .Site.GetPage "section" "home" -}} {{- if not $home -}}{{- $home = .Site.GetPage "/home" -}}{{- end -}} {{- if not $home -}}{{- $home = .Site.GetPage "page" "home/index" -}}{{- end -}} {{- if not $home -}}

위젯 섹션을 찾을 수 없습니다.

{{- else -}}
{{- /* 2) 위젯 수집: 섹션이면 .Pages, 리프면 .Resources */ -}} {{- $widgets := slice -}} {{- if eq $home.Kind "section" -}} {{- $widgets = $home.Pages -}} {{- else -}} {{- $widgets = $home.Resources.ByType "page" -}} {{- if not $widgets -}}{{- $widgets = $home.Resources.Match "**.md" -}}{{- end -}} {{- end -}} {{- /* widget 지정된 항목만 weight 오름차순 */ -}} {{- $widgets = where $widgets ".Params.widget" "!=" nil -}} {{- $widgets = sort $widgets "Params.weight" "asc" -}} {{- range $w := $widgets -}} {{- $wname := lower (printf "%v" (index $w.Params "widget")) -}} {{- $partial := printf "widgets/%s.html" (replace $wname "." "_") -}} {{- if templates.Exists (printf "partials/%s" $partial) -}} {{- partial $partial (dict "page" . "widget" $w) -}} {{- else -}}
알 수 없는 위젯: {{ $wname }}
{{- end -}} {{- end -}}
{{- end -}}
