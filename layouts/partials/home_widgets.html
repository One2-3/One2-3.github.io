"_" -}} {{/* about.avatar → about_avatar */}} {{- $partial := printf "widgets/%s.html" $wfile -}} {{- if templates.Exists (printf "partials/%s" $partial) -}} {{- partial $partial (dict "page" . "widget" $w) -}} {{- else -}}
