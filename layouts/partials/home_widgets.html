{{/* home_widgets.html — 다국어 안전판: GetPage 안 씀, 섹션 스캔 방식 */}} {{- $site := .Site -}} {{- $lang := $site.Language.Lang -}}
{{/* 1) 현재 언어 페이지 중 섹션이 home 인 페이지만 수집 */}} {{- $pages := where $site.RegularPages "Lang" $lang -}} {{- $widgets := where $pages "Section" "home" -}} {{/* 2) widget 프론트매터가 있는 것만 weight 오름차순 */}} {{- $widgets = where $widgets ".Params.widget" "!=" nil -}} {{- $widgets = sort $widgets "Params.weight" "asc" -}} {{- if not (gt (len $widgets) 0) -}}

위젯 섹션을 찾을 수 없습니다.

{{- else -}} {{- range $w := $widgets -}} {{- $name := lower (printf "%v" (index $w.Params "widget")) -}} {{- $partial := printf "widgets/%s.html" (replace $name "." "_") -}} {{- if templates.Exists (printf "partials/%s" $partial) -}} {{- partial $partial (dict "page" . "widget" $w) -}} {{- else -}}
Unknown widget: {{ $name }}
{{- end -}} {{- end -}} {{- end -}}
